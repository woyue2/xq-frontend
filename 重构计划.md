## 阶段 2：数据库层迁移（Week 1-2）

### 2.1 Prisma → Supabase JS SDK

| 当前 (Prisma) | 改为 (Supabase JS) |
|--------------|-------------------|
| `prisma.question.findMany()` | `supabase.from('Question').select('*')` |
| `prisma.$transaction([...])` | RPC Functions (复杂事务) |
| Raw SQL | `supabase.rpc('function_name', params)` |

### 2.2 RLS (Row Level Security) 策略

为每张表配置访问控制：

```sql
-- 示例: Question 表 RLS
alter table "Question" enable row level security;

-- 所有人可读已审核通过的问题
create policy "Public read approved questions"
  on "Question" for select
  using (status = 'approved');

-- 用户只能改自己的问题
create policy "Users update own questions"
  on "Question" for update
  using (auth.uid() = "authorId");
```

### 2.3 数据库 Functions (替代复杂事务)

```sql
-- 原: Prisma transaction 创建问题 + 扣减课时
-- 新: PostgreSQL Function
create or replace function create_question_with_hours(
  p_title text,
  p_content text,
  p_author_id uuid,
  p_subject_id int
) returns json as $$
declare
  v_question_id bigint;
  v_remaining_hours int;
begin
  -- 检查并扣减课时
  update "UserClassHours" 
  set "remainingHours" = "remainingHours" - 1,
      "consumedHours" = "consumedHours" + 1
  where "userId" = p_author_id and "remainingHours" > 0
  returning "remainingHours" into v_remaining_hours;
  
  if v_remaining_hours is null then
    raise exception 'Insufficient class hours';
  end if;
  
  -- 创建问题
  insert into "Question" (title, content, "authorId", "subjectId", status)
  values (p_title, p_content, p_author_id, p_subject_id, 'pending')
  returning id into v_question_id;
  
  return json_build_object(
    'questionId', v_question_id,
    'remainingHours', v_remaining_hours
  );
end;
$$ language plpgsql security definer;
```

### 3.2 前端鉴权 Store 重构

```typescript
// src/stores/useAuthStore.ts (新)
import { supabase } from '@/lib/supabase'

export const useAuthStore = create((set, get) => ({
  user: null,
  session: null,
  
  // 发送验证码
  async sendCode(phone: string) {
    const { error } = await supabase.auth.signInWithOtp({
      phone,
      options: {
        channel: 'sms'
      }
    })
    return { error }
  },
  
  // 验证码登录
  async verifyCode(phone: string, code: string) {
    const { data, error } = await supabase.auth.verifyOtp({
      phone,
      token: code,
      type: 'sms'
    })
    
    if (data.user) {
      // 检查白名单 (调用 Edge Function)
      const { data: whitelistCheck } = await supabase.functions.invoke(
        'check-whitelist',
        { body: { phone } }
      )
      
      if (!whitelistCheck.allowed) {
        await supabase.auth.signOut()
        throw new Error('手机号不在白名单中')
      }
    }
    
    return { data, error }
  },
  
  // 登出
  async logout() {
    await supabase.auth.signOut()
    set({ user: null, session: null })
  }
}))
```

  )
})

### 4.2 Vercel API Route 示例

```typescript
// api/questions/index.ts (Vercel Serverless Function)
import type { VercelRequest, VercelResponse } from '@vercel/node'
import { createClient } from '@supabase/supabase-js'

export default async function handler(
  req: VercelRequest,
  res: VercelResponse
) {
  const supabase = createClient(
    process.env.SUPABASE_URL!,
    process.env.SUPABASE_SERVICE_ROLE_KEY!
  )
  
  // 验证 JWT
  const token = req.headers.authorization?.replace('Bearer ', '')
  const { data: { user }, error: authError } = await supabase.auth.getUser(token)
  
  if (authError || !user) {
    return res.status(401).json({ error: 'Unauthorized' })
  }
  
  if (req.method === 'POST') {
    // 调用数据库 function
    const { data, error } = await supabase.rpc('create_question_with_hours', {
      p_title: req.body.title,
      p_content: req.body.content,
      p_author_id: user.id,
      p_subject_id: req.body.subjectId
    })
    
    if (error) {
      return res.status(400).json({ error: error.message })
    }
    
    // 触发 AI 审核 (异步)
    await supabase.functions.invoke('ai-audit', {
      body: { questionId: data.questionId }
    })
    
    return res.status(201).json(data)
  }
  
  if (req.method === 'GET') {
    const { data, error } = await supabase
      .from('Question')
      .select('*, author:User(id, nickname, avatarUrl)')
      .eq('status', 'approved')
      .order('createdAt', { ascending: false })
      .limit(20)
    
    return res.json(data)
  }
}
```

### 4.3 AI 审核 Edge Function

```typescript
// supabase/functions/ai-audit/index.ts
import { serve } from 'https://deno.land/std@0.177.0/http/server.ts'

serve(async (req) => {
  const { questionId } = await req.json()
  
  // 1. 获取问题内容
  const supabaseAdmin = createClient(
    Deno.env.get('SUPABASE_URL')!,
    Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
  )
  
  const { data: question } = await supabaseAdmin
    .from('Question')
    .select('title, content, images')
    .eq('id', questionId)
    .single()
  
  // 2. 调用 AI 审核
  const aiResult = await fetch('https://open.bigmodel.cn/api/paas/v4/chat/completions', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${Deno.env.get('AI_API_KEY')}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      model: 'glm-4-flash',
      messages: [
        { role: 'system', content: SYSTEM_PROMPT },
        { role: 'user', content: question.title + '\n' + question.content }
      ]
    })
  }).then(r => r.json())
  
  // 3. 解析结果并更新数据库
  const result = parseAiResponse(aiResult)
  
  await supabaseAdmin
    .from('Question')
    .update({
      status: result.safe ? 'approved' : 'rejected',
      aiAudit: result,
      aiAuditedAt: new Date().toISOString()
    })
    .eq('id', questionId)
  
  // 4. 发送通知 (如果违规)
  if (!result.safe) {
    await supabaseAdmin.functions.invoke('send-notification', {
      body: { userId: question.authorId, message: '内容违规：' + result.reason }
    })
  }
  
  return new Response('OK')
})
```

## 阶段 5：前端全面适配（Week 5-7）

### 5.1 API 调用层重构

```typescript
// src/services/api.ts (新)
import { supabase } from '@/lib/supabase'

export const questionApi = {
  // 列表 - 直查 Supabase
  async list(params?: ListParams) {
    let query = supabase
      .from('Question')
      .select('*, author:User(id, nickname, avatarUrl), answers:Answer(count)')
      .eq('status', 'approved')
    
    if (params?.subjectId) {
      query = query.eq('subjectId', params.subjectId)
    }
    
    const { data, error } = await query
      .order('createdAt', { ascending: false })
      .range(params?.offset || 0, (params?.offset || 0) + 19)
    
    if (error) throw error
    return data
  },
  
  // 创建 - 调 Vercel API (需要复杂事务)
  async create(data: CreateQuestionData) {
    const { data: { session } } = await supabase.auth.getSession()
    
    const res = await fetch('/api/questions', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${session?.access_token}`
      },
      body: JSON.stringify(data)
    })
    
    if (!res.ok) throw new Error(await res.text())
    return res.json()
  }
}
```

## 风险与应对

| 风险 | 可能性 | 应对方案 |
|------|--------|----------|
| RLS 配置错误导致数据泄露 | 🟡 中 | 每个策略编写后安全测试；使用 `security definer` 谨慎 |
| Edge Function 冷启动延迟 | 🟡 中 | AI 审核本来就是异步；关键路径用 Vercel API Routes |
| 复杂事务迁移困难 | 🟡 中 | 复杂逻辑保留为 PostgreSQL Functions |
| 老用户数据迁移 | 🟢 低 | Supabase Auth 用户 ID 与现有 User 表映射 |
