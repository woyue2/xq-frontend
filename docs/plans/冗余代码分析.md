### 7.4 后端降级抽象（P2）

#### 现状

每个服务都有类似的降级逻辑：

```typescript
if (process.env.NODE_ENV !== 'production') {
    coreLogger.warn({ mode: 'degraded', ... }, '...');
    // 降级处理
}
```

#### 优化方案

```typescript
// backend/src/utils/degradation.ts

export interface DegradationOptions {
  feature: string;
  reason: string;
  fallback?: () => Promise<any>;
}

export async function withDegradation<T>(
  options: DegradationOptions,
  productionLogic: () => Promise<T>,
): Promise<T> {
  try {
    return await productionLogic();
  } catch (err) {
    if (process.env.NODE_ENV === 'production') {
      throw err;
    }

    coreLogger.warn({ mode: 'degraded', ...options }, `Degraded: ${options.feature}`);

    if (options.fallback) {
      return await options.fallback();
    }

    throw err;
  }
}

// 使用
export class QuestionService {
  async create(params: CreateQuestionParams) {
    return withDegradation(
      {
        feature: 'question.create',
        reason: 'author not found',
        fallback: () => this.createFallbackUser(params),
      },
      () => this.createInternal(params),
    );
  }
}
```
